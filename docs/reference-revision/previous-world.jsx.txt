import { useEffect, useRef, useState } from 'react'
import { Link, useLocation, useNavigate } from 'react-router-dom'
import { gsap } from 'gsap'
import { usePortfolioUI } from '../../app/PortfolioUIContext.jsx'
import { worldDestinations, WORLD_ASSETS } from './world.config.js'
import './PortfolioWorld.css'

function IslandArt({ destination }) {
  const [failed, setFailed] = useState(false)
  return <>
    {!failed && <img className="pw-island-art" src={`${WORLD_ASSETS}${destination.id}.webp`} srcSet={`${WORLD_ASSETS}${destination.id}-640.webp 640w, ${WORLD_ASSETS}${destination.id}.webp 1200w`} sizes={destination.id === 'projects' ? '(max-width: 767px) 70vw, (max-width: 1023px) 51vw, 36vw' : '(max-width: 767px) 49vw, (max-width: 1023px) 34vw, 24vw'} alt="" width="1200" height="1200" decoding="async" onError={() => setFailed(true)} draggable="false" />}
    {failed && <span className="pw-art-fallback" aria-hidden="true">{destination.place}</span>}
    <span className="pw-waterfall pw-waterfall-one" aria-hidden="true"><i /><b /></span>
    {destination.id === 'projects' && <span className="pw-waterfall pw-waterfall-two" aria-hidden="true"><i /><b /></span>}
    <span className="pw-island-mist" aria-hidden="true" />
    <span className={`pw-landmark-life pw-landmark-${destination.id}`} aria-hidden="true" />
    {destination.id === 'about' && <span className="pw-canopy" aria-hidden="true" />}
    {(destination.id === 'qa' || destination.id === 'skills') && <span className="pw-mechanism" aria-hidden="true" />}
  </>
}

export default function PortfolioWorld() {
  const { reduced, tone } = usePortfolioUI()
  const location = useLocation()
  const navigate = useNavigate()
  const root = useRef(null)
  const camera = useRef(null)
  const timeline = useRef(null)
  const watchdog = useRef(null)
  const [selected, setSelected] = useState(() => {
    try { const saved = sessionStorage.getItem('portfolio-world-selection'); return worldDestinations.some(item => item.id === saved) ? saved : 'projects' } catch { return 'projects' }
  })
  const [arriving, setArriving] = useState(Boolean(location.state?.arrival) && !reduced)
  const [entering, setEntering] = useState(null)
  const [hidden, setHidden] = useState(document.hidden)
  const destination = worldDestinations.find(item => item.id === selected)
  useEffect(() => {
    try { sessionStorage.setItem('portfolio-world-selection', selected) } catch { /* optional */ }
  }, [selected])
  useEffect(() => {
    if (!arriving) return
    navigate('/world-map', { replace: true, state: null })
    const timer = setTimeout(() => setArriving(false), 1400)
    return () => clearTimeout(timer)
  }, [arriving, navigate])
  useEffect(() => {
    const visibility = () => setHidden(document.hidden)
    document.addEventListener('visibilitychange', visibility)
    return () => { document.removeEventListener('visibilitychange', visibility); timeline.current?.kill(); clearTimeout(watchdog.current) }
  }, [])
  useEffect(() => {
    if (!reduced || !entering) return
    timeline.current?.kill(); clearTimeout(watchdog.current); navigate(entering.route)
  }, [reduced, entering, navigate])
  useEffect(() => {
    if (!root.current || reduced) return
    const element = root.current
    const finePointer = matchMedia('(hover: hover) and (pointer: fine)')
    const move = event => {
      if (!finePointer.matches || entering || document.hidden) return
      const bounds = element.getBoundingClientRect()
      element.style.setProperty('--pointer-x', `${((event.clientX - bounds.left) / bounds.width - .5) * 10}px`)
      element.style.setProperty('--pointer-y', `${((event.clientY - bounds.top) / bounds.height - .5) * 6}px`)
    }
    const reset = () => { element.style.setProperty('--pointer-x', '0px'); element.style.setProperty('--pointer-y', '0px') }
    element.addEventListener('pointermove', move, { passive: true }); element.addEventListener('pointerleave', reset)
    return () => { element.removeEventListener('pointermove', move); element.removeEventListener('pointerleave', reset); reset() }
  }, [reduced, entering])
  function enter(event, item) {
    if (event.button !== 0 || event.metaKey || event.ctrlKey || event.shiftKey || event.altKey || reduced) return
    event.preventDefault()
    if (entering) return
    setSelected(item.id); setEntering(item); tone()
    const finish = () => navigate(item.route)
    watchdog.current = setTimeout(finish, 1900)
    try {
      const island = root.current.querySelector(`[data-island="${item.id}"]`)
      const bounds = camera.current.getBoundingClientRect(), target = island.getBoundingClientRect()
      const originX = target.left + target.width / 2 - bounds.left, originY = target.top + target.height * .38 - bounds.top
      timeline.current = gsap.timeline({ onComplete: () => { clearTimeout(watchdog.current); finish() } })
        .to(root.current.querySelector('.pw-character'), { y: -8, rotation: item.x < 50 ? -4 : 4, duration: .22 })
        .to(camera.current, { transformOrigin: `${originX}px ${originY}px`, scale: 3.3, x: (bounds.width / 2 - originX) * .5, y: (bounds.height / 2 - originY) * .35, duration: 1.15, ease: 'power2.inOut' }, .12)
        .to(root.current.querySelector('.pw-foreground'), { scale: 1.5, y: 160, opacity: 0, duration: .65 }, .15)
        .fromTo(root.current.querySelector('.pw-entry-cloud'), { xPercent: -110, opacity: 0 }, { xPercent: 0, opacity: 1, duration: .7, ease: 'power2.in' }, .55)
    } catch { clearTimeout(watchdog.current); finish() }
  }
  return (
    <main id="main" ref={root} className={`portfolio-world ${entering ? 'is-entering' : ''} ${arriving && !reduced ? 'is-arriving' : ''}`} data-paused={reduced || hidden} data-looking={destination.x < 50 ? 'left' : 'right'}>
      <div className="pw-sky" aria-hidden="true" /><div className="pw-sunlight" aria-hidden="true" />
      <div className="pw-world-heading"><p className="pw-eyebrow">JUNYOUNG KIM · PORTFOLIO</p><h1 tabIndex="-1">Same me.<br /><span>A bigger world.</span></h1><p>작은 호기심이, 더 넓은 경험으로.</p></div>
      <div className="pw-camera" ref={camera}>
        <div className="pw-far-islands" aria-hidden="true"><i /><i /><i /><i /></div>
        <div className="pw-clouds pw-clouds-far" aria-hidden="true" />
        <div className="pw-depth-plane"><nav className="pw-islands" aria-label="세계의 다섯 목적지">
          {worldDestinations.map(item => <Link key={item.id} to={item.route} className={`pw-island pw-island-${item.id} ${selected === item.id ? 'is-selected' : ''}`} data-island={item.id}
            style={{ '--island-x': `${item.x}%`, '--island-y': `${item.y}%`, '--island-width': `${item.width}%`, '--float-duration': `${item.duration}s`, '--float-phase': `${item.phase}s`, '--float-drift': `${item.drift}px` }}
            onPointerEnter={() => !entering && setSelected(item.id)} onFocus={() => !entering && setSelected(item.id)} onClick={event => enter(event, item)} aria-label={`${item.title} — ${item.subtitle} 탐색`}>
            <span className="pw-island-float"><span className="pw-island-response"><IslandArt destination={item} /></span></span>
            <span className="pw-island-label"><strong>{item.title}</strong><span>{item.subtitle}</span><i aria-hidden="true">↗</i></span>
          </Link>)}
        </nav></div>
        <div className="pw-clouds pw-clouds-mid" aria-hidden="true" /><div className="pw-clouds pw-cloud-sea" aria-hidden="true" />
        <div className="pw-birds" aria-hidden="true"><i /><i /><i /></div><div className="pw-airship" aria-hidden="true"><i /></div>
      </div>
      <div className="pw-foreground" aria-hidden="true"><div className="pw-lookout" /><div className="pw-character"><div className="pw-character-idle"><i className="pw-pose pw-pose-base" /><i className="pw-pose pw-pose-left" /><i className="pw-pose pw-pose-ready" /></div></div>
        <div className="pw-grasses">{Array.from({ length: 12 }, (_, i) => <i key={i} style={{ '--blade': i }} />)}</div><div className="pw-leaves"><i /><i /><i /></div>
      </div>
      <aside className="pw-discovery" aria-label="선택한 목적지"><span className="pw-eyebrow">EXPLORE THE WORLD</span><h2>{destination.place}</h2><p>{destination.description}</p><Link to={destination.route} onClick={event => enter(event, destination)}>{destination.title} 둘러보기 <span aria-hidden="true">↗</span></Link></aside>
      <div className="pw-world-footer"><span>다섯 개의 장소, 하나의 이야기.</span><Link to="/projects">프로젝트 바로 보기 ↗</Link></div>
      <div className="pw-entry-cloud" aria-hidden="true" />
      {entering && <div className="pw-transition-status"><span role="status">{entering.title}에 도착하는 중</span><Link to={entering.route}>바로 이동 ↗</Link></div>}
    </main>
  )
}
